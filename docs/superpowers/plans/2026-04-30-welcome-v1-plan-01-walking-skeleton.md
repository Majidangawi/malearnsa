# Welcome /welcome v1 — Plan 1: Walking Skeleton

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Render a personalized hero greeting in Saudi Arabic on `staging-welcome.malearnsa.com` for a test JWT representing "Khalid using Midjourney" — proving the edge middleware, JWT decode, signal bundle, LLM call, output validation, KV cache, and Server Component render path all work end-to-end on the hidden staging environment.

**Architecture:** Brand-new Next.js 15 (App Router) repo deployed to Vercel Pro with deployment-level password protection. Vercel Edge Middleware decodes signed JWT + cookies + referrer/UTM into a signal bundle, looks up Vercel KV cache, calls Anthropic Haiku 4.5 on miss, validates JSON output, and hands off to a streaming Server Component that renders the hero. Supabase `staging` schema logs every visit. No tip cards, no chat, no captures, no dashboard yet — those land in Plans 2-7.

**Tech Stack:**
- Next.js 15 (App Router, React Server Components, streaming)
- TypeScript (strict mode)
- Vercel Pro (deployment password + KV + Edge runtime)
- Anthropic SDK (`@anthropic-ai/sdk`) with Claude Haiku 4.5 (`claude-haiku-4-5-20251001`)
- Supabase (existing `malearn-chat` project, new `staging` schema)
- Vitest for unit tests
- Playwright for one end-to-end smoke test
- `jose` for JWT verify (HS256, shared secret)

**Repo:** `Majidangawi/ma-learn-welcome` (new). Pushed via Noor's classic PAT per locked feedback memory.

**End state of Plan 1 (demo criteria):** Visit `staging-welcome.malearnsa.com/welcome?ref=ig_bio&t=<test_jwt>` after entering the Vercel password. Page renders within 1.5s on Saudi 4G mobile. Hero displays a Saudi-Arabic-dialect greeting that references Midjourney by name, generated by Haiku 4.5 on cache miss or served from KV on cache hit. `staging.welcome_visits` table contains a row with the full signal bundle. Search engines blocked at three layers.

```typescript
// lib/signals/jwt.ts
import { jwtVerify } from "jose";
import type { ManyChatJwtPayload, EmailJwtPayload } from "@/lib/types";

type VerifyResult<T> =
  | { ok: true; payload: T }
  | {
      ok: false;
      reason: "expired" | "invalid_signature" | "missing_fields" | "missing_secret" | "malformed";
    };

function getSecret(envVar: string): Uint8Array | null {
  const value = process.env[envVar];
  if (!value) return null;
  return new TextEncoder().encode(value);
}

async function verifyJwt<T>(
  token: string,
  envVar: string,
  requiredFields: readonly string[]
): Promise<VerifyResult<T>> {
  const secret = getSecret(envVar);
  if (!secret) return { ok: false, reason: "missing_secret" };

  try {
    const { payload } = await jwtVerify(token, secret, { algorithms: ["HS256"] });
    for (const field of requiredFields) {
      if (!(field in payload)) return { ok: false, reason: "missing_fields" };
    }
    return { ok: true, payload: payload as T };
  } catch (e) {
    const msg = e instanceof Error ? e.message : String(e);
    if (msg.includes("expired")) return { ok: false, reason: "expired" };
    if (msg.includes("signature")) return { ok: false, reason: "invalid_signature" };
    return { ok: false, reason: "malformed" };
  }
}

export function verifyManyChatJwt(token: string) {
  return verifyJwt<ManyChatJwtPayload>(token, "MANYCHAT_JWT_SECRET", ["sub", "ig_handle"]);
}

export function verifyEmailJwt(token: string) {
  return verifyJwt<EmailJwtPayload>(token, "EMAIL_JWT_SECRET", ["sub", "email"]);
}
```

```typescript
// lib/signals/bundle.ts
import type {
  DayPart,
  Language,
  ManyChatJwtPayload,
  EmailJwtPayload,
  SignalBundle,
  SourceChannel,
  ToolId,
  ToolSignal,
} from "@/lib/types";
import { evaluateLeadState } from "@/lib/signals/leadState";

const REFERRER_TO_TOOL: Record<string, ToolId> = {
  "midjourney.com": "midjourney",
  "higgsfield.ai": "higgsfield",
  "weavy.ai": "weavy",
  "magnific.ai": "magnific",
  "lumalabs.ai": "luma",
  "luma.ai": "luma",
  "openart.ai": "openart",
};

export interface SignalBundleInputs {
  now: Date;
  url: URL;
  cookie: { cookie_id: string } | null;
  manyChatPayload: ManyChatJwtPayload | null;
  emailPayload: EmailJwtPayload | null;
  acceptLanguageHeader: string;
  referrer: string | null;
  isCustomer: boolean;
}

function detectLanguage(header: string): Language {
  const first = header.split(",")[0]?.split(";")[0]?.toLowerCase() ?? "";
  return first.startsWith("en") ? "en" : "ar";
}

function detectDayPart(now: Date): DayPart {
  // Use KSA timezone (UTC+3) consistently
  const ksaHours = (now.getUTCHours() + 3) % 24;
  if (ksaHours >= 5 && ksaHours < 12) return "morning";
  if (ksaHours >= 12 && ksaHours < 17) return "afternoon";
  if (ksaHours >= 17 && ksaHours < 22) return "evening";
  return "night";
}

function detectToolFromReferrer(referrer: string | null): ToolId | null {
  if (!referrer) return null;
  try {
    const host = new URL(referrer).hostname.toLowerCase();
    for (const [domain, tool] of Object.entries(REFERRER_TO_TOOL)) {
      if (host === domain || host.endsWith(`.${domain}`)) return tool;
    }
  } catch {
    return null;
  }
  return null;
}

function detectSourceChannel(url: URL, hasManyChatPayload: boolean, hasEmailPayload: boolean): SourceChannel {
  const ref = url.searchParams.get("ref");
  if (ref === "ig_bio") return "ig_bio";
  if (ref === "manychat" || hasManyChatPayload) return "manychat";
  if (ref === "email" || hasEmailPayload) return "email";
  if (ref === "ad") return "ad";
  if (ref) return "direct";
  return "direct";
}

export function buildSignalBundle(inputs: SignalBundleInputs): SignalBundle {
  const language = detectLanguage(inputs.acceptLanguageHeader);
  const day_part = detectDayPart(inputs.now);

  let toolSignal: ToolSignal = { source: "none", tool: null, confidence: 0 };
  if (inputs.manyChatPayload) {
    // ManyChat doesn't pass tool today; reserved for v2 tool-tagged JWTs
    toolSignal = { source: "manychat", tool: null, confidence: 0.7 };
  }
  const refTool = detectToolFromReferrer(inputs.referrer);
  if (refTool) {
    toolSignal = { source: "referrer", tool: refTool, confidence: 0.9 };
  }

  const name =
    inputs.manyChatPayload?.name ??
    inputs.emailPayload?.name ??
    null;
  const ig_handle = inputs.manyChatPayload?.ig_handle ?? null;
  const email = inputs.emailPayload?.email ?? null;
  const returning = inputs.cookie !== null;

  const lead_state = evaluateLeadState({
    is_customer: inputs.isCustomer,
    returning,
    has_tool_signal: toolSignal.tool !== null,
    from_email: inputs.emailPayload !== null,
    from_manychat: inputs.manyChatPayload !== null,
  });

  return {
    language,
    name,
    ig_handle,
    email,
    tool_signal: toolSignal,
    lead_state,
    returning,
    is_customer: inputs.isCustomer,
    day_part,
    source_channel: detectSourceChannel(inputs.url, !!inputs.manyChatPayload, !!inputs.emailPayload),
  };
}
```

```typescript
// lib/llm/validate.ts
import type { GreetingOutput, CtaTier, Language } from "@/lib/types";

const VALID_TIERS: CtaTier[] = ["T1", "T2", "T3", "T4"];
const VALID_LANGS: Language[] = ["ar", "en"];

const BANNED_PHRASES_LOWERCASE = [
  "as an ai",
  "i am an ai",
  "i'm an ai",
  "i am here to help",
  "feel free to",
  "i'd be happy to",
  "kindly note",
  "as a language model",
  "أنا ذكاء اصطناعي",
  "أنا برنامج",
];

const MAX_WORDS_GREETING = 25;
const MAX_WORDS_CLOSE = 35;

type ValidateResult =
  | { ok: true; output: GreetingOutput }
  | {
      ok: false;
      reason:
        | "malformed_json"
        | "missing_fields"
        | "invalid_language"
        | "invalid_tier"
        | "length_exceeded"
        | "banned_phrase";
    };

function countWords(s: string): number {
  return s.trim().split(/\s+/).filter(Boolean).length;
}

function containsBannedPhrase(s: string): boolean {
  const lower = s.toLowerCase();
  return BANNED_PHRASES_LOWERCASE.some((p) => lower.includes(p));
}

export function validateGreetingOutput(raw: string): ValidateResult {
  let parsed: unknown;
  try {
    parsed = JSON.parse(raw);
  } catch {
    return { ok: false, reason: "malformed_json" };
  }

  if (typeof parsed !== "object" || parsed === null) {
    return { ok: false, reason: "missing_fields" };
  }
  const o = parsed as Record<string, unknown>;

  if (typeof o.greeting_line !== "string" || typeof o.close_line !== "string") {
    return { ok: false, reason: "missing_fields" };
  }
  if (!Array.isArray(o.tip_card_ids)) return { ok: false, reason: "missing_fields" };
  if (typeof o.recommended_cta !== "object" || o.recommended_cta === null)
    return { ok: false, reason: "missing_fields" };

  const cta = o.recommended_cta as Record<string, unknown>;
  if (typeof cta.tier !== "string" || !VALID_TIERS.includes(cta.tier as CtaTier))
    return { ok: false, reason: "invalid_tier" };
  if (typeof cta.reason !== "string") return { ok: false, reason: "missing_fields" };

  if (typeof o.language !== "string" || !VALID_LANGS.includes(o.language as Language))
    return { ok: false, reason: "invalid_language" };

  if (countWords(o.greeting_line) > MAX_WORDS_GREETING)
    return { ok: false, reason: "length_exceeded" };
  if (countWords(o.close_line) > MAX_WORDS_CLOSE)
    return { ok: false, reason: "length_exceeded" };

  if (containsBannedPhrase(o.greeting_line) || containsBannedPhrase(o.close_line))
    return { ok: false, reason: "banned_phrase" };

  return {
    ok: true,
    output: {
      greeting_line: o.greeting_line,
      language: o.language as Language,
      tip_card_ids: (o.tip_card_ids as unknown[]).filter((x): x is string => typeof x === "string"),
      recommended_cta: { tier: cta.tier as CtaTier, reason: cta.reason },
      close_line: o.close_line,
    },
  };
}
```

```typescript
// lib/llm/anthropic.ts
import Anthropic from "@anthropic-ai/sdk";
import type { GreetingOutput, SignalBundle } from "@/lib/types";
import { buildSystemPrompt, buildUserMessage } from "@/lib/llm/prompts";
import { validateGreetingOutput } from "@/lib/llm/validate";

const MODEL_ID = "claude-haiku-4-5-20251001";
const MAX_TOKENS = 400;
const TIMEOUT_MS = 800;

export type GreetingGenerationResult =
  | { ok: true; output: GreetingOutput; latencyMs: number }
  | { ok: false; reason: string; latencyMs: number };

export async function generateGreeting(
  bundle: SignalBundle
): Promise<GreetingGenerationResult> {
  const apiKey = process.env.ANTHROPIC_API_KEY;
  if (!apiKey) {
    return { ok: false, reason: "missing_api_key", latencyMs: 0 };
  }
  const client = new Anthropic({ apiKey });
  const start = Date.now();

  try {
    const response = await Promise.race([
      client.messages.create({
        model: MODEL_ID,
        max_tokens: MAX_TOKENS,
        system: buildSystemPrompt(bundle.language),
        messages: [{ role: "user", content: buildUserMessage(bundle) }],
        metadata: { user_id: bundle.ig_handle ?? bundle.email ?? "anonymous" },
      }),
      new Promise<never>((_, reject) =>
        setTimeout(() => reject(new Error("timeout")), TIMEOUT_MS)
      ),
    ]);

    const latencyMs = Date.now() - start;
    const block = response.content[0];
    if (!block || block.type !== "text") {
      return { ok: false, reason: "no_text_block", latencyMs };
    }
    const validated = validateGreetingOutput(block.text);
    if (!validated.ok) {
      return { ok: false, reason: `validation:${validated.reason}`, latencyMs };
    }
    return { ok: true, output: validated.output, latencyMs };
  } catch (e) {
    const latencyMs = Date.now() - start;
    const msg = e instanceof Error ? e.message : String(e);
    return { ok: false, reason: `api:${msg}`, latencyMs };
  }
}
```

```typescript
// lib/cache/kv.ts
import { kv } from "@vercel/kv";
import type { GreetingOutput, SignalBundle } from "@/lib/types";

const TTL_SECONDS = 60 * 60 * 24; // 24 hours

export function greetingCacheKey(
  bundle: SignalBundle,
  promptVersion: number
): string {
  // Bucket dimensions: tool, lead_state, language, day_part, prompt_version
  // Name/handle/email/customer_id are NOT in the bucket — many users share the same greeting
  const tool = bundle.tool_signal.tool ?? "none";
  return `g:v${promptVersion}:${tool}:${bundle.lead_state}:${bundle.language}:${bundle.day_part}`;
}

export async function getCachedGreeting(
  bundle: SignalBundle,
  promptVersion: number
): Promise<GreetingOutput | null> {
  const key = greetingCacheKey(bundle, promptVersion);
  try {
    const cached = await kv.get<GreetingOutput>(key);
    return cached ?? null;
  } catch {
    return null;
  }
}

export async function setCachedGreeting(
  bundle: SignalBundle,
  promptVersion: number,
  output: GreetingOutput
): Promise<void> {
  const key = greetingCacheKey(bundle, promptVersion);
  try {
    await kv.set(key, output, { ex: TTL_SECONDS });
  } catch {
    // Cache failures are non-fatal
  }
}
```

```typescript
// middleware.ts
import { NextRequest, NextResponse } from "next/server";
import { verifyManyChatJwt, verifyEmailJwt } from "@/lib/signals/jwt";
import { buildSignalBundle } from "@/lib/signals/bundle";
import { generateGreeting } from "@/lib/llm/anthropic";
import { staticToolFallback, genericFallback } from "@/lib/llm/fallback";
import { getCachedGreeting, setCachedGreeting } from "@/lib/cache/kv";
import { GREETING_GENERATOR_VERSION } from "@/lib/llm/prompts";
import { logVisit } from "@/lib/db/visits";
import type { GreetingSource } from "@/lib/types";

export const config = {
  matcher: ["/welcome", "/welcome/:path*"],
  runtime: "nodejs", // node runtime needed for Anthropic SDK on edge
};

const COOKIE_NAME = "ml_w_cid";

function readOrSetCookieId(req: NextRequest): { cookie_id: string; isNew: boolean } {
  const existing = req.cookies.get(COOKIE_NAME)?.value;
  if (existing) return { cookie_id: existing, isNew: false };
  const newId = crypto.randomUUID();
  return { cookie_id: newId, isNew: true };
}

export async function middleware(req: NextRequest) {
  const url = new URL(req.url);
  const tokenParam = url.searchParams.get("t");

  let manyChatPayload = null;
  let emailPayload = null;

  if (tokenParam) {
    const ref = url.searchParams.get("ref");
    if (ref === "ig_bio" || ref === "manychat") {
      const result = await verifyManyChatJwt(tokenParam);
      if (result.ok) manyChatPayload = result.payload;
    } else if (ref === "email") {
      const result = await verifyEmailJwt(tokenParam);
      if (result.ok) emailPayload = result.payload;
    }
  }

  const { cookie_id, isNew } = readOrSetCookieId(req);

  const bundle = buildSignalBundle({
    now: new Date(),
    url,
    cookie: isNew ? null : { cookie_id },
    manyChatPayload,
    emailPayload,
    acceptLanguageHeader: req.headers.get("accept-language") ?? "ar",
    referrer: req.headers.get("referer"),
    isCustomer: false, // wired in Plan 2 once customers table is read
  });

  let output = await getCachedGreeting(bundle, GREETING_GENERATOR_VERSION);
  let source: GreetingSource = "cache";

  if (!output) {
    const result = await generateGreeting(bundle);
    if (result.ok) {
      output = result.output;
      source = "llm";
      // fire-and-forget cache write
      void setCachedGreeting(bundle, GREETING_GENERATOR_VERSION, result.output);
    } else if (bundle.tool_signal.tool) {
      output = staticToolFallback(bundle.tool_signal.tool, bundle.language);
      source = "static_fallback";
    } else {
      output = genericFallback(bundle.language);
      source = "generic_fallback";
    }
  }

  // fire-and-forget visit log
  void logVisit({
    cookie_id,
    signal_bundle: bundle,
    greeting_source: source,
    output,
  });

  // Pass output + bundle to the page via request headers (read by Server Component)
  const response = NextResponse.next({
    request: {
      headers: new Headers({
        ...Object.fromEntries(req.headers),
        "x-greeting": JSON.stringify(output),
        "x-signal-bundle": JSON.stringify(bundle),
        "x-greeting-source": source,
      }),
    },
  });

  if (isNew) {
    response.cookies.set(COOKIE_NAME, cookie_id, {
      httpOnly: true,
      secure: true,
      sameSite: "lax",
      maxAge: 60 * 60 * 24 * 90, // 90 days, matches retention policy
    });
  }

  return response;
}
```
